Bayesian updating: the globe toss

Choose a prior, add tosses, and watch prior and likelihood combine into the posterior.

What proportion \(\theta\) of the Earth’s surface is water? Pick a prior on the left, set the data, and watch the three curves of Bayes’ theorem respond. This page goes with week 6, the logic of Bayesian inference.

A preset fills in the two shape parameters of a Beta prior. Their ratio sets where it sits, and their sum sets how sure it is. The data are tosses of a simulated globe whose true proportion of water you choose.

viewof reset = Inputs.button("Reset to defaults")

presets = ({
  "Uniform": { a: 1, b: 1 },
  "Weakly informative": { a: 2, b: 2 },
  "Informative": { a: 14, b: 6 },
  "Dogmatic": { a: 500, b: 500 }
})

viewof preset = (reset, Inputs.select(Object.keys(presets), {

Things to try

Predict first, then move the control.

  1. The defaults (10 water in 15 tosses), uniform prior, then informative prior. Where does the posterior sit relative to the prior and the likelihood?
  2. Weakly informative prior, then slide the tosses from 1 to 1000. What happens to the prior’s influence, and to the width of the posterior?
  3. Dogmatic prior with the true \(\theta\) at 0.71. How many tosses does it take before the posterior leaves 0.5?
  4. Rule out \(\theta\) below 0.75 and keep the true \(\theta\) at 0.71. What does the posterior do as the tosses grow, and why can it never reach the truth?

Simulated tosses are added to one fixed sequence, so moving the slider right means observing more of the same globe, and moving it left takes tosses back. New sequence starts a fresh one, and Reset to defaults restores every control. The cutoff toggle sets the prior to exactly zero below the cutoff: a belief held with certainty, which no amount of data can revive.

How the curves are computed

The page uses exactly the four ingredients from the session, on a grid of 1000 values of \(\theta\). At each grid value, prior times likelihood is computed, and dividing by the sum of all of them, the normalising constant, turns the result into a posterior that sums to 1.

\[ P(\theta_i \mid \text{data}) = \frac{P(\text{data} \mid \theta_i) \, P(\theta_i)}{\sum_{j} P(\text{data} \mid \theta_j) \, P(\theta_j)} \tag{1}\]

Equation 1 is Bayes’ theorem with the integral in the denominator replaced by a sum over the grid. The likelihood is binomial, so \(P(\text{data} \mid \theta) \propto \theta^{W} (1 - \theta)^{N - W}\), and the binomial coefficient cancels in the division.

function mulberry32(seed) {
  return function () {
    seed |= 0
    seed = (seed + 0x6d2b79f5) | 0
    let t = Math.imul(seed ^ (seed >>> 15), 1 | seed)
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
  }
}

// One fixed sequence of 1000 uniforms per seed. Toss i is water when its
// uniform falls below the true theta, so more tosses only ever add to it.
// Seed 3595 gives 10 water in the first 15 tosses at theta = 0.71, the deck's example.
uniforms = {
  const rng = mulberry32(3595 + redraw)
  return Float64Array.from({ length: 1000 }, rng)
}

data = {
  const N = Math.round(nSim)
  let W = 0
  for (let i = 0; i < N; i++) if (uniforms[i] < trueTheta) W++
  return { W, N }
}
// 1000 grid points at the cell midpoints, so 0, 0.5 and 1 are never on the grid.
grid = {
  const G = 1000
  const d = 1 / G
  const theta = Array.from({ length: G }, (_, i) => (i + 0.5) * d)

  // Log space, so N = 1000 cannot underflow.
  const logPrior = theta.map((t) =>
    zeroOn && t < cutoff ? -Infinity : (a - 1) * Math.log(t) + (b - 1) * Math.log(1 - t)
  )
  const logLik = theta.map((t) => data.W * Math.log(t) + (data.N - data.W) * Math.log(1 - t))
  const logPost = logPrior.map((lp, i) => lp + logLik[i])

  // Subtract the maximum, exponentiate, divide by the sum: the normalising constant.
  const toWeights = (logw) => {
    const m = Math.max(...logw.filter(Number.isFinite))
    const w = logw.map((x) => Math.exp(x - m))
    const s = w.reduce((acc, x) => acc + x, 0)
    return w.map((x) => x / s)
  }
  const prior = toWeights(logPrior)
  const post = toWeights(logPost)
  const likMax = Math.max(...logLik)
  const lik = logLik.map((x) => Math.exp(x - likMax))

  // Posterior summaries.
  let cum = 0
  let lo = null
  let hi = null
  let mean = 0
  let above = 0
  for (let i = 0; i < G; i++) {
    cum += post[i]
    mean += theta[i] * post[i]
    if (theta[i] > 0.5) above += post[i]
    if (lo === null && cum >= 0.05) lo = theta[i]
    if (hi === null && cum >= 0.95) hi = theta[i]
  }

  const rows = theta.map((t, i) => ({
    theta: t,
    prior: prior[i] / d,
    lik: lik[i],
    post: post[i] / d
  }))
  return { rows, mean, lo, hi, above }
}
panel = (y, label, colour, extra = []) =>
  Plot.plot({
    width: panelWidth,
    height: panelHeight + (y === "post" ? 14 : 0),
    style: { fontSize: "13px" },
    marginLeft: 55,
    marginBottom: y === "post" ? 42 : 28,
    x: {
      domain: [0, 1],
      label: y === "post" ? "θ, proportion of water" : null,
      labelAnchor: "center",
      labelArrow: "none"
    },
    y: { label, grid: true },
    marks: [
      Plot.areaY(grid.rows, { x: "theta", y, fill: colour, fillOpacity: 0.15 }),
      ...extra,
      Plot.lineY(grid.rows, { x: "theta", y, stroke: colour, strokeWidth: 2 }),
      Plot.ruleX([data.W / data.N], { stroke: "currentColor", strokeOpacity: 0.6 }),
      Plot.ruleX([trueTheta], { stroke: "currentColor", strokeDasharray: "4,3" }),
      Plot.ruleY([0])
    ]
  })

// The three stacked panels: prior, likelihood, posterior with its 90 percent interval.
panels = () => html`<div>
  ${panel("prior", "Prior (density)", "steelblue")}
  ${panel("lik", "Likelihood (rescaled)", "#CD9B1D")}
  ${panel("post", "Posterior (density)", "#E07B39", [
    Plot.areaY(
      grid.rows.filter((r) => r.theta >= grid.lo && r.theta <= grid.hi),
      { x: "theta", y: "post", fill: "#E07B39", fillOpacity: 0.35 }
    )
  ])}
</div>`

fmt = (x) => x.toFixed(3)

fmtProb = (p) => (p < 0.0005 ? "< 0.001" : p > 0.9995 ? "> 0.999" : p.toFixed(3))
